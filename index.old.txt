import express, { Application, Request, Response } from "express";
import mongoose, { Schema, Document, Model } from "mongoose";
import cors from "cors";
import dotenv from "dotenv";

const app: Application = express();
dotenv.config();

app.use(
  cors({
    origin: [
      "http://localhost:5173",
      "https://ramadan-bags.vercel.app",
      "https://ramadan-bags.pages.dev",
    ],
    credentials: true,
    optionsSuccessStatus: 200,
  }),
);
app.use(express.json());
app.use(express.urlencoded({ extended: true }));

// Validate environment variables
if (!process.env.MDBUSER || !process.env.MDBPWD || !process.env.MDBDB) {
  console.error("Missing MongoDB credentials");
  process.exit(1);
}

const mongoURI = `mongodb+srv://${process.env.MDBUSER}:${process.env.MDBPWD}@cluster0.iumas.mongodb.net/${process.env.MDBDB}?retryWrites=true&w=majority`;

// Connect with modern options (avoid deprecation warnings)
mongoose
  .connect(mongoURI, {
    serverSelectionTimeoutMS: 5000,
  })
  .then(() => console.log("MongoDB Connected"))
  .catch((err) => {
    console.error("MongoDB Connection Error:", err);
    process.exit(1);
  });

interface IitemSchema extends Document {
  name: string;
  total: number;
  unit: string;
  available: number;
}

const itemSchema: Schema = new mongoose.Schema({
  name: { type: String, required: true },
  total: { type: Number, required: true },
  unit: { type: String, required: true },
  available: { type: Number, required: true },
});

const Item: Model<IitemSchema> = mongoose.model<IitemSchema>(
  "Item",
  itemSchema,
);

app.get("/", async (req: Request, res: Response) => {
  return res.status(200).send("ok");
});

app.post("/login", async (req: Request, res: Response) => {
  try {
    if (req.body.pw === process.env.AdminPw) {
      return res.status(200).json({ user: "admin" });
    } else {
      return res.status(401).json();
    }
  } catch {
    return res.status(500).json();
  }
});

app.get("/items", async (req: Request, res: Response) => {
  try {
    const items = await Item.find().lean();
    return res.status(200).json({ items });
  } catch (err) {
    console.error(err);
    return res.status(500).json();
  }
});

app.post("/items", async (req: Request, res: Response) => {
  try {
    await Item.insertMany(req.body);
    const allItems = await Item.find().lean();
    return res.status(201).json({ items: allItems });
  } catch (err) {
    console.error(err);
    return res.status(500).json();
  }
});

app.put("/items", async (req: Request, res: Response) => {
  try {
    const bulkOps = req.body.map((item: IitemSchema) => ({
      updateOne: {
        filter: { _id: item._id },
        update: { $set: item },
      },
    }));
    await Item.bulkWrite(bulkOps);
    const allItems = await Item.find().lean();
    return res.status(200).json({ items: allItems });
  } catch (err) {
    console.error(err);
    return res.status(500).json();
  }
});

app.delete("/items", async (req: Request, res: Response) => {
  try {
    const bulkOps = req.body.map((id: string) => ({
      deleteOne: {
        filter: { _id: id },
      },
    }));
    await Item.bulkWrite(bulkOps);
    const allItems = await Item.find().lean();
    return res.status(200).json({ items: allItems });
  } catch (err) {
    console.error(err);
    return res.status(500).json();
  }
});

// Graceful shutdown
process.on("SIGINT", async () => {
  await mongoose.disconnect();
  console.log("MongoDB disconnected");
  process.exit(0);
});

app.listen(3005, () => console.log(`Server running on port 3005`));
